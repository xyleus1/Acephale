'use strict';
const fs = require('node:fs');
const path = require('node:path');
const {execFileSync} = require('node:child_process');
const c = require('./composition-core.cjs');

function writeExclusive(relative, data) {
  const target=path.join(c.ROOT,relative);
  fs.mkdirSync(path.dirname(target),{recursive:true});
  fs.writeFileSync(target,data,{flag:'wx'});
}
function main() {
  const retainedOnly=process.argv.length===3 && process.argv[2]==='--validate-retained';
  c.check(process.argv.length===2 || retainedOnly,'Only --validate-retained is accepted.');
  c.frozen();
  const locks=c.readJson(path.join(c.ROOT,'input-lock.json'));
  c.check(locks.sourceCommit===c.COMMIT,'Wrong input lock source.');
  const excludedRaw=c.sourceFile(path.join(c.ROOT,'excluded-run-ids.json'),locks);
  const excluded=JSON.parse(excludedRaw.data);
  c.check(excluded.database==='derya_main_benchmark_resume' && excluded.profile==='graph-large' && excluded.count===45 && new Set(excluded.runIds).size===45,'Wrong excluded interrupted run list.');
  const inputs=(retainedOnly?c.SPECS.slice(0,2):c.SPECS).map(spec=>c.loadProfile(spec,locks,new Set(excluded.runIds)));
  c.check(new Set(inputs.map(p=>p.audit.fingerprint)).size===1 && new Set(inputs.map(p=>p.audit.runtime)).size===1,'Source fingerprint/runtime changed across accepted profiles.');
  c.frozen();
  if(retainedOnly) { console.log(JSON.stringify({ok:true,mode:'retained-profile-audit',profiles:inputs.map(p=>({profile:p.spec.profile,runs:p.audit.runs,executions:p.audit.executions})),databaseWrites:false},null,2)); return; }
  for(const relative of ['results','sources','profile-provenance.json','composition-status.json']) c.check(!fs.existsSync(path.join(c.ROOT,relative)),`Refusing to overwrite existing composition: ${relative}`);
  const headers=inputs.map(p=>p.csv.slice(0,p.csv.indexOf('\n')+1));
  c.check(new Set(headers).size===1 && headers[0].length>0 && inputs.every(p=>p.csv.endsWith('\n')),'CSV headers or final newlines differ.');
  const summary={schemaVersion:1,generatedAt:new Date().toISOString()};
  for(const key of ['configurations','systems','runs','bandwidthMultiplier','projectionAmortization']) {
    c.check(inputs.every(p=>Array.isArray(p.summary[key])),`Missing summary ${key}.`);
    summary[key]=inputs.flatMap(p=>p.summary[key]);
  }
  c.check(summary.runs.length===150 && new Set(summary.runs.map(r=>r.runId)).size===150,'Incomplete or duplicate combined runs.');
  const copies=new Map();
  for(const input of inputs) {
    for(const [name,raw] of Object.entries(input.artifacts)) copies.set(`sources/${input.spec.profile}/${name}`,raw);
    const sourceName=input.spec.sourceRoot===c.RESUME?'resume':'large';
    copies.set(`sources/${sourceName}/cohort-status.json`,input.statusRaw);
    copies.set(`sources/${sourceName}/run-cohort.ps1`,input.launcherRaw);
  }
  const provenance={schemaVersion:1,kind:'complete_profile_composition',sourceCommit:c.COMMIT,sourceTreeFingerprint:inputs[0].audit.fingerprint,
    expectedRuns:150,expectedExecutions:30000,interruptedRunIdsExcluded:excluded.runIds,excludedRunListSha256:excludedRaw.sha256,
    publicationScope:'Two complete profiles retained from a launch interrupted only later; entire 250K profile rerun independently. Not one uninterrupted cohort and no partial-pass splicing.',
    profiles:inputs.map(input=>({...input.audit,profileArtifacts:Object.fromEntries(Object.entries(input.artifacts).map(([name,raw])=>[name,{path:`sources/${input.spec.profile}/${name}`,sha256:raw.sha256,bytes:raw.bytes}])),
      sourceStatusPath:`sources/${input.spec.sourceRoot===c.RESUME?'resume':'large'}/cohort-status.json`,sourceLauncherPath:`sources/${input.spec.sourceRoot===c.RESUME?'resume':'large'}/run-cohort.ps1`})),
    originalArtifacts:[...copies].map(([relative,raw])=>({path:relative,sha256:raw.sha256,bytes:raw.bytes}))};
  for(const [relative,raw] of copies) writeExclusive(relative,raw.data);
  writeExclusive('results/results.csv',headers[0]+inputs.map(p=>p.csv.slice(p.csv.indexOf('\n')+1)).join(''));
  writeExclusive('results/summary.json',JSON.stringify(summary,null,2)+'\n');
  writeExclusive('profile-provenance.json',JSON.stringify(provenance,null,2)+'\n');
  execFileSync(process.execPath,['--import','tsx',path.join(c.ROOT,'render-composed-report.cjs')],{cwd:c.REPO,windowsHide:true,stdio:'pipe'});
  c.reportIds(fs.readFileSync(path.join(c.ROOT,'results/report.html'),'utf8'),summary);
  c.frozen();
  const artifacts=['results/results.csv','results/summary.json','results/report.html','profile-provenance.json'].map(relative=>{const bytes=fs.readFileSync(path.join(c.ROOT,relative));return{path:relative,sha256:c.hash(bytes),bytes:bytes.length};});
  writeExclusive('composition-status.json',JSON.stringify({status:'completed',kind:'complete_profile_composition',sourceCommit:c.COMMIT,runs:150,executions:30000,completedAt:new Date().toISOString(),artifacts},null,2)+'\n');
  console.log(JSON.stringify({ok:true,runs:150,executions:30000,profileComposition:true,output:c.ROOT},null,2));
}
main();
