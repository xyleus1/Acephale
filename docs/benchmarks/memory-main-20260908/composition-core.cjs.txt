function check(condition, message) { if (!condition) throw new Error(message); }
function hash(bytes) { return crypto.createHash('sha256').update(bytes).digest('hex'); }
function readJson(file) { return JSON.parse(fs.readFileSync(file, 'utf8').replace(/^\uFEFF/, '')); }
function stable(value) {
  if (Array.isArray(value)) return `[${value.map(stable).join(',')}]`;
  if (value && typeof value === 'object') return `{${Object.keys(value).sort().map(k => `${JSON.stringify(k)}:${stable(value[k])}`).join(',')}}`;
  return JSON.stringify(value);
}

  check(!/-----BEGIN [A-Z ]*PRIVATE KEY-----/.test(text), 'Private key in artifact.');

function parseCsv(text) {
  const records = [], fields = []; let value = '', quoted = false, afterQuote = false;
  for (let index = 0; index < text.length; index++) {
    const char = text[index];
    if (quoted) {
      if (char === '"' && text[index+1] === '"') { value += '"'; index++; }
      else if (char === '"') { quoted = false; afterQuote = true; }
      else value += char;
    } else if (char === '"' && value === '' && !afterQuote) quoted = true;
    else if (char === ',') { fields.push(value); value = ''; afterQuote = false; }
    else if (char === '\n' || char === '\r') {
      if (char === '\r' && text[index+1] === '\n') index++;
      fields.push(value); records.push(fields.splice(0)); value = ''; afterQuote = false;
    } else { check(!afterQuote, 'Malformed CSV after closing quote.'); value += char; }
  }
  check(!quoted, 'Unclosed CSV quoted field.');
  if (value !== '' || fields.length || afterQuote) { fields.push(value); records.push(fields); }
  const headers = records.shift(); check(headers && new Set(headers).size === headers.length, 'Missing/duplicate CSV headers.');
  const rows = records.map(record => { check(record.length === headers.length, 'Wrong CSV field count.'); return Object.fromEntries(headers.map((h,i)=>[h,record[i]])); });
  return {headers, rows};
}
function assertProfile(summary, csv, status, spec, deniedIds) {
  const infrastructure = assertInfrastructure(status, spec);
  check(summary.schemaVersion === 1 && summary.runs?.length === 50 && summary.configurations?.length === 50 && summary.systems?.length === 5, 'Incomplete profile summary.');
  const {rows,headers} = typeof csv === 'string' ? parseCsv(csv) : csv;
  for (const name of ['run_id','status','failure_code','profile','system_name','case_id','measurement_pass','query_ordinal','tenant_leak_count','unrouted_leak_count']) check(headers.includes(name), `Missing ${name} CSV column.`);
  check(rows.length === 10000, 'Profile must contain exactly 10,000 result rows.');
  const byId = new Map(), grid = new Set(), configTimes = new Set(), vectors = new Set(), runtime = new Set(), fingerprints = new Set();
  const windowStart = timestamp(infrastructure.stage.startedAt), windowEnd = timestamp(infrastructure.stage.completedAt);
  for (const run of summary.runs) {
    const c = run.configuration;
    check(run.profile === spec.profile && SYSTEMS.includes(run.systemName) && c?.status === 'completed' && c.profile === spec.profile && c.databaseLabel === spec.database, 'Wrong run profile/system/status/database.');
    check(c.commitSha === COMMIT && c.workingTreeDirty === false && /^[0-9a-f]{64}$/.test(c.sourceTreeFingerprint), 'Invalid frozen source provenance.');
    for (const [key,wanted] of Object.entries({corpusSize:spec.size,queryCount:200,topK:20,runCount:10,warmupPassCount:3,concurrency:4,queryOrderSeed:42,postgresStatementTimeout:'10min',queryOrderPolicy:'seeded_random_per_pass_shared_across_systems',warmupPolicy:'full_corpus_seeded_random'})) check(c[key] === wanted, `Unexpected profile setting ${key}.`);
    check(Number.isInteger(c.runIndex) && c.runIndex >= 1 && c.runIndex <= 10, 'Wrong measured pass index.');
    const cell = `${run.systemName}:${c.runIndex}`;
    check(!grid.has(cell), 'Duplicate system/pass.'); grid.add(cell);
    check(typeof run.runId === 'string' && run.runId.startsWith(`benchmark:${spec.profile}:${run.systemName.toLowerCase()}:${c.runIndex}:`) && !byId.has(run.runId) && !deniedIds.has(run.runId), 'Duplicate, malformed, or excluded interrupted run ID.');
    byId.set(run.runId, run);
    const runStart = timestamp(run.startedAt), runEnd = timestamp(run.completedAt), configured = timestamp(c.timestamp);
    check(configured >= windowStart && configured <= windowEnd && runStart >= windowStart && runEnd >= runStart && runEnd <= windowEnd, 'Run/configuration outside selected launch window.');
    const v = c.vectorIndex;
    check(v?.profile === spec.profile && v.modelId === 'Xenova/all-MiniLM-L6-v2' && v.modelRevision === 'cb3d680149bf9a3209564e1b27ab3bb355b65707' && v.modelDtype === 'q8' && v.dimensions === 384 && v.runtimeVersion === '4.2.0' && v.corpusFingerprint === spec.corpus && v.indexChunkCount === spec.chunks, 'Wrong pinned vector model or corpus.');
    check(v.modelAssets?.length > 0 && [...v.modelAssets,v.vectorAsset,v.metadataAsset].every(a => Number.isFinite(a?.bytes) && a.bytes > 0 && /^[0-9a-f]{64}$/.test(a.sha256)), 'Missing vector asset hashes.');
    fingerprints.add(c.sourceTreeFingerprint); configTimes.add(c.timestamp); vectors.add(stable(v));
    runtime.add(stable(Object.fromEntries(['branch','commitSha','sourceTreeFingerprint','sourceTreeFingerprintAlgorithm','sourceTreeFileCount','hardwareLabel','runtimeLabel','postgresVersion','postgresStatementTimeout','prismaPoolConfiguration','embeddingModel','embeddingDimensions'].map(k=>[k,c[k]]))));
  }
  check(grid.size === 50 && configTimes.size === 1 && vectors.size === 1 && fingerprints.size === 1 && runtime.size === 1, 'Mixed launch/source/runtime configuration inside profile.');
  check(stable(summary.configurations.map(stable).sort()) === stable(summary.runs.map(r=>stable(r.configuration)).sort()), 'Top-level configurations differ from measured runs.');
  sameSet(summary.systems.map(s=>s.systemName), SYSTEMS, 'summary system set');
  for (const system of summary.systems) {
    check(system.profile === spec.profile && system.status === 'completed' && system.metrics && system.configuration?.databaseLabel === spec.database, 'Incomplete system summary.');
    sameSet(system.measuredRunIds, summary.runs.filter(r=>r.systemName===system.systemName).map(r=>r.runId), 'system measured run IDs');
    check(Array.isArray(system.warmups) && system.warmups.length === 1, 'Missing unique warmup record.');
    const warmup = system.warmups[0];
    check(warmup.passCount === 3 && warmup.executionCount === 600 && warmup.failureCount === 0 && new Set(warmup.queryOrderHashes).size === 3, 'Wrong warmups or warmup failures.');
  }
  const countByRun = new Map(), cells = new Set(), cases = new Map(), orders = new Map();
  for (const row of rows) {
    const run = byId.get(row.run_id);
    check(run && !deniedIds.has(row.run_id) && row.status === 'completed' && row.failure_code === '' && row.profile === spec.profile && row.system_name === run.systemName, 'Unexpected, failed, or excluded CSV row.');
    check(Number(row.measurement_pass) === run.configuration.runIndex && Number.isInteger(Number(row.query_ordinal)) && Number(row.query_ordinal) >= 1 && Number(row.query_ordinal) <= 200 && row.case_id !== '', 'Invalid case/pass/query ordinal.');
    check(row.tenant_leak_count === '0' && row.unrouted_leak_count === '0', 'Leakage or missing leakage measurement.');
    const cell = `${row.run_id}:${row.case_id}`; check(!cells.has(cell), 'Duplicate query row.'); cells.add(cell);
    countByRun.set(row.run_id,(countByRun.get(row.run_id)||0)+1);
    if (!cases.has(row.run_id)) cases.set(row.run_id,new Set()); cases.get(row.run_id).add(row.case_id);
    if (!orders.has(row.run_id)) orders.set(row.run_id,[]); orders.get(row.run_id).push([Number(row.query_ordinal),row.case_id]);
  }
  const referenceCases = [...cases.values()][0];
  const passOrders = new Map(), passHashes = new Map();
  for (const run of summary.runs) {
    check(countByRun.get(run.runId) === 200 && cases.get(run.runId).size === 200 && [...referenceCases].every(id=>cases.get(run.runId).has(id)), 'Incomplete or nonmatching case set.');
    const order = orders.get(run.runId).sort((a,b)=>a[0]-b[0]);
    check(order.every(([ordinal],i)=>ordinal===i+1), 'Duplicate or missing query ordinal.');
    const signature = stable(order), pass = run.configuration.runIndex;
    if (passOrders.has(pass)) check(passOrders.get(pass)===signature && passHashes.get(pass)===run.configuration.queryOrderHash, 'Systems have different order within a pass.');
    else { passOrders.set(pass,signature); passHashes.set(pass,run.configuration.queryOrderHash); }
  }
  check(new Set(passOrders.values()).size === 10, 'Pass query orders were not independently randomized.');
  check(timestamp(summary.generatedAt) >= Math.max(...summary.runs.map(r=>timestamp(r.completedAt))) && timestamp(summary.generatedAt) <= windowEnd, 'Profile summary generation outside completed stage.');
  return {profile:spec.profile,database:spec.database,sourceRoot:spec.sourceRoot,...infrastructure,runs:50,executions:10000,runIds:[...byId.keys()],fingerprint:[...fingerprints][0],runtime:[...runtime][0]};
}
function reportIds(html, summary) {
  const observed = [...html.matchAll(/<td>(benchmark:[^<]+)<\/td>/g)].map(m=>m[1]);
  sameSet(observed, summary.runs.map(r=>r.runId), 'HTML report run IDs');
}

function sourceFile(file, locks) {
  const data = fs.readFileSync(file), digest = hash(data), locked = locks.artifacts.find(a=>a.path.toLowerCase()===file.toLowerCase());
  if (locked) check(locked.sha256 === digest && locked.bytes === data.length, `Frozen input changed: ${file}`);
  return {data,sha256:digest,bytes:data.length};
}
function loadProfile(spec, locks, denied) {
  const statusFile = path.join(spec.sourceRoot,'cohort-status.json');
  const launcherFile = path.join(spec.sourceRoot,'run-cohort.ps1');
  const statusRaw = sourceFile(statusFile,locks), launcherRaw = sourceFile(launcherFile,locks);
  const artifacts = {};
  for (const name of ['summary.json','results.csv','report.html']) artifacts[name] = sourceFile(path.join(spec.sourceRoot,'profiles',spec.profile,name),locks);
  const status=JSON.parse(statusRaw.data), summary=JSON.parse(artifacts['summary.json'].data), csv=artifacts['results.csv'].data.toString('utf8');
  for (const value of [...Object.values(artifacts),statusRaw,launcherRaw]) secretScan(value.data.toString('utf8'));
  const audit=assertProfile(summary,csv,status,spec,denied); reportIds(artifacts['report.html'].data.toString('utf8'),summary);
  return {spec,status,summary,csv,artifacts,statusRaw,launcherRaw,audit};
}
