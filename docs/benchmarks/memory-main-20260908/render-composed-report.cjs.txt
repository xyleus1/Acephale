'use strict';
// Pure HTML rendering from already audited profile summaries; no database imports.
const fs=require('node:fs');
const path=require('node:path');
const {createRequire}=require('node:module');
const c=require('./composition-core.cjs');
c.frozen();
const source=createRequire(path.join(c.REPO,'package.json'));
const {renderBenchmarkReport}=source('./src/lib/supply-os/benchmark/report.ts');
const summary=c.readJson(path.join(c.ROOT,'results/summary.json'));
const notice='<aside style="padding:16px;border:1px solid #94a3b8;margin:16px"><strong>Profile-composed benchmark.</strong> These are three complete profiles from two launches, not one uninterrupted cohort. Every interrupted 250K run is excluded. Legacy action-safety labels below refer to seeded single-dependency validation fixtures and modeled no-validator controls, not actual sends or post-read unrelated mutations. See profile-provenance.json and analysis/findings.md for scope and limitations.</aside>';
const html=renderBenchmarkReport(summary).replace(/(<body[^>]*>)/i,`$1${notice}`);
c.reportIds(html,summary);
fs.writeFileSync(path.join(c.ROOT,'results/report.html'),html,{encoding:'utf8',flag:'wx'});
